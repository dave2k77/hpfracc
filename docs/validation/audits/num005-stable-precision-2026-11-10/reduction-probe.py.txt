"""Controlled float32 history-reduction probes for NUM-005."""

import hashlib
import json
from pathlib import Path

import jax
import jax.numpy as jnp
import numpy as np

import hpfracc as hp
import hpfracc.solvers.predictor_corrector as pc
from benchmarks.numerical.implicit_validation import independent_residuals


def model(t, state, params, *, rng_key=None, inputs=None):
    return -params * state


def tree_sum(weights, history):
    shape = (weights.shape[0],) + (1,) * (history.ndim - 1)
    return jnp.sum(weights.reshape(shape) * history, axis=0)


def highest_dot(weights, history):
    return jnp.tensordot(weights, history, axes=([0], [0]), precision="highest")


original = pc._history_dot
rows = []
with jax.enable_x64(False):
    times = jnp.linspace(0., 1., 257, dtype=jnp.float32)
    initial = jnp.ones(1, dtype=jnp.float32)
    order = jnp.asarray(.5, dtype=jnp.float32)
    rate = jnp.asarray(30., dtype=jnp.float32)
    for label, reduction in (("default_dot", original),
                             ("highest_dot", highest_dot),
                             ("elementwise_sum", tree_sum)):
        try:
            pc._history_dot = reduction
            for mode in ("stable", "direct"):
                result = hp.solvers.simulate(
                    model=model, ts=times, initial_state=initial, params=rate,
                    solver=hp.solvers.ImplicitPredictorCorrector(
                        dt=1/256, order=order, newton_iterations=8,
                        newton_rtol=1e-5, newton_atol=1e-7,
                        weight_evaluation=mode,
                    ),
                )
                assert result.latent_state.dtype == jnp.float32
                assert result.solver_info is not None
                residuals = independent_residuals(
                    "half_decay", np.asarray(times),
                    np.asarray(result.latent_state, dtype=np.float64),
                    30., .5, np.ones(1, dtype=np.float64), 1e-5, 1e-7,
                )
                rows.append({"reduction": label, "weight_evaluation": mode,
                             "independent_max_scaled_residual": float(np.max(residuals)),
                             "worst_step": int(np.argmax(residuals)) + 1,
                             "reported_convergence": bool(
                                 result.solver_info.diagnostics["newton_converged"]),
                             "dtype": str(result.latent_state.dtype)})
        finally:
            pc._history_dot = original
print(json.dumps({
    "scope": "History-reduction counterfactuals; no production change or IVP accuracy claim",
    "backend": jax.default_backend(), "jax": jax.__version__,
    "source_sha256": hashlib.sha256(Path(pc.__file__).read_bytes()).hexdigest(),
    "rows": rows,
}, indent=2))
